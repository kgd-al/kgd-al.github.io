% Colours.  ink = tree/branch colour, acc = life-form colour.
% One palette for every background.  ink is a mid-dark green: contrast ratio about 2.7:1
% on a dark tab strip (#202124), 6:1 on white, and 1.8:1 against acc, so the branches stay
% readable on both and still stand apart from the objects.
\definecolor{ink}{HTML}{16723A}
\definecolor{acc}{HTML}{16A34A}
\tikzset{
  glyph/.style={fill=acc,draw=none},
  glyphl/.style={draw=acc,line width=1.1pt,line cap=round},
}

%% ================= temporal radial tree =================
% radius = time since LUCA (centre).  All tips sit on the same circle (today).
% \split{parent r}{angle1}{r1}{angle2}{r2}: arc at the parent's radius, then radial branches.
\newcommand{\split}[5]{%
  \draw (#2:#1) arc[start angle=#2,end angle=#4,radius=#1];
  \draw (#2:#1) -- (#2:#3);
  \draw (#4:#1) -- (#4:#5);}

% 7 tips. Bacteria split first (right after LUCA); then plants, then fungi, then
% jellyfish, then fish; robot and glider branch off last (short, recent twigs).
% Tips are every 360/7 deg; internal nodes sit at the mean angle of their children.
% Leaf order around the circle: bacteria, seedling, jellyfish, robot, glider, fish, mushroom
\newcommand{\treeSeven}[2]{% R (tip radius), node-dot radius
  \pgfmathsetmacro{\rrA}{0.30*#1}\pgfmathsetmacro{\rrB}{0.50*#1}
  \pgfmathsetmacro{\rrC}{0.66*#1}\pgfmathsetmacro{\rrD}{0.78*#1}
  \pgfmathsetmacro{\rrE}{0.88*#1}
  \draw (0,0) -- (0:#1);                              % bacteria: first branch off LUCA
  \draw (0,0) -- (143.04:\rrA);                       % everything else
  \split{\rrA}{51.43}{#1}{234.64}{\rrB}               % seedling | (fungi+animals)
  \split{\rrB}{160.72}{\rrC}{308.57}{#1}              % animals | mushroom
  \split{\rrC}{102.86}{#1}{218.57}{\rrD}              % jellyfish | (fish + artificial)
  \split{\rrD}{180}{\rrE}{257.14}{#1}                 % artificial | fish
  \split{\rrE}{154.29}{#1}{205.71}{#1}                % robot | glider
  \pgfmathsetmacro{\rrR}{2.2*#2}
  \fill[acc] (0,0) circle (\rrR);                      % LUCA, the centre
  \fill (143.04:\rrA) circle (#2) (234.64:\rrB) circle (#2)
        (160.72:\rrC) circle (#2) (218.57:\rrD) circle (#2) (180:\rrE) circle (#2);}

% 7 tips for tiny sizes: same order, but only 3 rings so that the arcs stay resolvable.
% Splits that are close together in \treeSeven are merged (jellyfish/fish/robot/glider
% branch from shared nodes), which keeps the order bacteria < plants < mushroom < the rest.
\newcommand{\treeSevenS}[2]{% R (tip radius), LUCA-dot radius
  \pgfmathsetmacro{\rrA}{0.42*#1}\pgfmathsetmacro{\rrB}{0.68*#1}\pgfmathsetmacro{\rrD}{0.90*#1}
  \draw (0,0) -- (0:#1);                              % bacteria
  \draw (0,0) -- (128.57:\rrA);
  \split{\rrA}{51.43}{#1}{205.71}{\rrB}               % plants | rest
  \split{\rrB}{102.86}{#1}{308.57}{#1}                % jellyfish | ... | mushroom
  \draw (205.71:\rrB) -- (205.71:\rrD);
  \draw (154.29:\rrD) arc[start angle=154.29,end angle=257.14,radius=\rrD];
  \draw (154.29:\rrD) -- (154.29:#1) (205.71:\rrD) -- (205.71:#1) (257.14:\rrD) -- (257.14:#1);
  \fill[acc] (0,0) circle (#2);}

% 4 tips, same logic, for tiny sizes: bacteria | plants | animals | artificial
\newcommand{\treeFour}[2]{%
  \pgfmathsetmacro{\rrA}{0.32*#1}\pgfmathsetmacro{\rrB}{0.68*#1}
  \draw (0,0) -- (0:#1);
  \draw (0,0) -- (157.5:\rrA);
  \split{\rrA}{90}{#1}{225}{\rrB}
  \split{\rrB}{180}{#1}{270}{#1}
  \pgfmathsetmacro{\rrR}{2.2*#2}
  \fill[acc] (0,0) circle (\rrR);                      % LUCA, the centre
  \fill (157.5:\rrA) circle (#2) (225:\rrB) circle (#2);}

%% ================= abstract shapes: edge count = age =================
% Branching order (oldest first): LUCA (circle, 0 edges) - bacteria (capsule) - plants
% (lens, 2 curved edges) - fungi (triangle) - jellyfish (square) - fish (pentagon) -
% robot + glider (hexagon; sister tips = same age = same shape).
% Each shape is drawn in a local frame: origin = attachment point (a vertex), +y = outward.
\newcommand{\aBact}{\fill[glyph] (-0.23,0.23) arc[start angle=180,end angle=360,radius=0.23]
  -- (0.23,0.77) arc[start angle=0,end angle=180,radius=0.23] -- cycle;}
\newcommand{\aLeaf}{\fill[glyph] (0,0) .. controls (0.42,0.12) and (0.42,0.88) .. (0,1)
  .. controls (-0.42,0.88) and (-0.42,0.12) .. (0,0);}
% regular n-gon of circumradius 0.5, one vertex on the origin, pointing inwards
\newcommand{\pn}[1]{\fill[glyph] (0,0) \foreach \k in {1,...,#1}{
  -- ({0.5*sin(\k*360/#1)},{0.5-0.5*cos(\k*360/#1)}) } -- cycle;}
\newcommand{\aTri}{\pn{3}}\newcommand{\aSqu}{\pn{4}}
\newcommand{\aPen}{\pn{5}}\newcommand{\aHex}{\pn{6}}

%% ================= how glyphs are attached to the branch tips =================
% \anch{name}{ax}{ay}{rot}{scale}: draw glyph \g<name> with its anchor point (ax,ay)
% moved onto the origin, rotated by rot degrees about that point, then scaled.
\newcommand{\anch}[5]{\begin{scope}[rotate=#4,scale=#5,shift={(-#2,-#3)}]\csname g#1\endcsname\end{scope}}
% \tipat{a}{R}{body}: origin = branch tip (tree angle a, radius R), y-axis = radially outwards,
% so that whatever #3 draws grows out of the end of the line.
% (The drawing is mirrored vertically and rotated by \aoff, so global angle = -(a+aoff).)
\newcommand{\tipat}[3]{%
  \pgfmathsetmacro{\tipang}{-(#1+\aoff)}\pgfmathsetmacro{\tiprot}{\tipang-90}%
  \begin{scope}[shift={(\tipang:#2)},rotate=\tiprot]#3\end{scope}}

%% ================= three resolutions =================
\newcommand{\aoff}{270}   % global rotation of the whole drawing (deg)
